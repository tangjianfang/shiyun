```js
const CARD_VARIANTS = new Set(['poem', 'stat', 'cloud-chip', 'cloud-node', 'skeleton']);

export function Card({ title, body, footer, variant = 'poem', onClick = '', className = '' } = {}) {
  if (body == null) throw new Error('Card: body is required');
  const v = CARD_VARIANTS.has(variant) ? variant : 'poem';
  const safeTitle = title != null ? `<h3 class="card__title">${esc(title)}</h3>` : '';
  const safeFooter = footer != null ? `<div class="card__footer">${footer}</div>` : '';
  const cls = `card card--${v}${className ? ' ' + className : ''}`;
  const clickAttr = onClick ? ` onclick="${onClick}"` : '';
  return `<div class="${cls}" data-variant="${v}"${clickAttr}>${safeTitle}<div class="card__body">${body}</div>${safeFooter}</div>`;
}
```

```js
export function EmptyState({ icon: ic = '', title, message = '', action = '' } = {}) {
  if (!title) throw new Error('EmptyState: title is required');
  const iconHtml = ic ? `<div class="empty-state__icon">${ic}</div>` : '';
  const actionHtml = action ? `<div class="empty-state__action">${action}</div>` : '';
  const msgHtml = message ? `<div class="empty-state__message">${esc(message)}</div>` : '';
  return `<div class="empty-state"><div class="empty-state__title">${esc(title)}</div>${iconHtml}${msgHtml}${actionHtml}</div>`;
}

// 向后兼容(下个 wave 移除)
export function emptyState(opts) {
  return EmptyState({ icon: opts.icon, title: opts.title, message: opts.body, action: opts.action });
}
```

```js
import { describe, it, expect, beforeEach } from 'vitest';
import { readFileSync, readdirSync, statSync } from 'fs';
import { join } from 'path';

function collectHtmlFiles(dir) {
  const out = [];
  for (const name of readdirSync(dir)) {
    const p = join(dir, name);
    if (statSync(p).isDirectory()) out.push(...collectHtmlFiles(p));
    else if (name.endsWith('.html')) out.push(p);
  }
  return out;
}

describe('语义层级', () => {
  it('学习版主页(home/learn)只有 1 个 <h1>', () => {
    const files = ['src/learning.template.html'];
    for (const f of files) {
      const html = readFileSync(f, 'utf8');
      const h1Count = (html.match(/<h1[\s>]/g) || []).length;
      expect(h1Count, f).toBe(1);
    }
  });

  it('所有 h2 必须出现在 h1 之后(松散检查)', () => {
    const html = readFileSync('src/learning.template.html', 'utf8');
    const h1Idx = html.search(/<h1[\s>]/);
    const h2Idx = html.search(/<h2[\s>]/);
    if (h2Idx >= 0) {
      expect(h1Idx).toBeLessThan(h2Idx);
    }
  });

  it('h3 不应出现在 h1/h2 缺失的孤立块中(教学一致性)', () => {
    // 静态扫描:每个 h3 的前面 200 字符内必须出现 h1 或 h2
    const html = readFileSync('src/learning.template.html', 'utf8');
    const h3Matches = [...html.matchAll(/<h3[\s>][^<]*<\/h3>/g)];
    for (const m of h3Matches) {
      const before = html.slice(Math.max(0, m.index - 200), m.index);
      const hasHeading = /<h[12][\s>]/.test(before);
      expect(hasHeading, `孤立 h3 at ${m.index}: ${m[0]}`).toBe(true);
    }
  });

  it('不出现 h4/h5/h6(降级扁平化)', () => {
    const html = readFileSync('src/learning.template.html', 'utf8');
    expect(html).not.toMatch(/<h4[\s>]/);
    expect(html).not.toMatch(/<h5[\s>]/);
    expect(html).not.toMatch(/<h6[\s>]/);
  });
});
```

```js
const FILTER_TYPES = new Set(['chip', 'checkbox', 'radio']);

function renderChipSection(sec) {
  const isActive = (v) => String(sec.value) === String(v);
  return sec.options.map(opt => `
    <button type="button" class="filter-chip${isActive(opt.value) ? ' filter-chip--active' : ''}"
      data-key="${esc(sec.key)}" data-value="${esc(opt.value)}"
      aria-pressed="${isActive(opt.value)}"
      onclick="${sec.onChange}('${esc(opt.value)}')">
      ${esc(opt.label)}${opt.count != null ? ` <span class="filter-chip__count">${opt.count}</span>` : ''}
    </button>
  `).join('');
}

function renderCheckboxSection(sec) {
  const values = sec.value || [];
  return sec.options.map(opt => {
    const checked = values.includes(opt.value);
    return `<label class="filter-checkbox${checked ? ' filter-checkbox--active' : ''}">
      <input type="checkbox" data-key="${esc(sec.key)}" data-value="${esc(opt.value)}"
        ${checked ? 'checked' : ''} onchange="${sec.onChange}('${esc(opt.value)}', this.checked)" />
      ${esc(opt.label)}${opt.count != null ? ` (${opt.count})` : ''}
    </label>`;
  }).join('');
}

function renderRadioSection(sec) {
  return sec.options.map(opt => `
    <label class="filter-radio${String(sec.value) === String(opt.value) ? ' filter-radio--active' : ''}">
      <input type="radio" name="filter-${esc(sec.key)}" data-key="${esc(sec.key)}" data-value="${esc(opt.value)}"
        ${String(sec.value) === String(opt.value) ? 'checked' : ''}
        onchange="${sec.onChange}('${esc(opt.value)}')" />
      ${esc(opt.label)}
    </label>
  `).join('');
}

export function FilterBar({ sections = [], collapsible = false, collapsedSummary = '筛选 ▾' } = {}) {
  const sectionsHtml = sections.map(sec => {
    if (!FILTER_TYPES.has(sec.type)) sec.type = 'chip';
    let body = '';
    if (sec.type === 'chip') body = renderChipSection(sec);
    else if (sec.type === 'checkbox') body = renderCheckboxSection(sec);
    else body = renderRadioSection(sec);
    const collapsed = sec.defaultCollapsed ? ' filter-section--collapsed' : '';
    return `<div class="filter-section${collapsed}" data-filter-type="${sec.type}">
      <div class="filter-section__label">${esc(sec.label)}</div>
      <div class="filter-section__body">${body}</div>
    </div>`;
  }).join('');
  const summary = collapsible ? `<div class="filter-bar__summary">${esc(collapsedSummary)}</div>` : '';
  return `<div class="filter-bar" data-collapsible="${collapsible}">${summary}${sectionsHtml}</div>`;
}
```

### Step 1.6.3: 实现 VirtualList

创建 `src/js/ui/virtual-list.js`:

```js
import { EmptyState } from './components.js';

function renderAllFallback(items, renderItem, errCb) {
  return items.map((it, i) => {
    try { return `<div class="virtual-list__item" data-index="${i}">${renderItem(it, i)}</div>`; }
    catch (e) { errCb(e, i); return `<div class="virtual-list__item virtual-list__item--error" data-index="${i}">${EmptyState({ title: '渲染失败', message: String(e.message) })}</div>`; }
  }).join('');
}

export function VirtualList({ items = [], renderItem, estimatedItemHeight = 40, overscan = 5, containerHeight, fallback = 'render-all' } = {}) {
  let rootEl = null;
  let observer = null;
  let state = { items: items.slice(), scrollEl: null, start: 0, end: 0 };
  const isSupported = typeof globalThis.IntersectionObserver === 'function';

  function computeVisible() {
    if (!state.scrollEl) return;
    const scrollTop = state.scrollEl.scrollTop;
    const h = estimatedItemHeight;
    const viewCount = Math.ceil((containerHeight || state.scrollEl.clientHeight) / h);
    state.start = Math.max(0, Math.floor(scrollTop / h) - overscan);
    state.end = Math.min(state.items.length, state.start + viewCount + 2 * overscan);
  }

  function renderVisible() {
    const slice = state.items.slice(state.start, state.end);
    const top = state.start * estimatedItemHeight;
    const html = slice.map((it, i) => {
      try { return `<div class="virtual-list__item" data-index="${state.start + i}" style="height:${estimatedItemHeight}px">${renderItem(it, state.start + i)}</div>`; }
      catch (e) { return `<div class="virtual-list__item virtual-list__item--error" data-index="${state.start + i}" style="height:${estimatedItemHeight}px">${EmptyState({ title: '渲染失败', message: String(e.message) })}</div>`; }
    }).join('');
    const layer = rootEl.querySelector('.virtual-list__layer');
    layer.style.transform = `translateY(${top}px)`;
    layer.innerHTML = html;
  }

  function mount(el) {
    state.scrollEl = el;
    rootEl = document.createElement('div');
    rootEl.className = 'virtual-list';
    const totalH = state.items.length * estimatedItemHeight;
    if (!isSupported || fallback === 'render-all') {
      rootEl.setAttribute('data-virtual-fallback', 'true');
      rootEl.innerHTML = renderAllFallback(state.items, renderItem, () => {});
      el.appendChild(rootEl);
      return;
    }
    rootEl.style.height = `${totalH}px`;
    rootEl.style.position = 'relative';
    rootEl.innerHTML = `<div class="virtual-list__sentinel" style="position:absolute;top:0;left:0;width:1px;height:1px"></div><div class="virtual-list__layer" style="position:absolute;top:0;left:0;width:100%"></div>`;
    el.appendChild(rootEl);
    computeVisible();
    renderVisible();
    observer = new IntersectionObserver(() => { computeVisible(); renderVisible(); }, { root: el, rootMargin: `${overscan * estimatedItemHeight}px` });
    observer.observe(rootEl.querySelector('.virtual-list__sentinel'));
    el.addEventListener('scroll', () => { computeVisible(); renderVisible(); });
  }

  function update(newItems) {
    state.items = newItems.slice();
    if (!isSupported || fallback === 'render-all') {
      rootEl.innerHTML = renderAllFallback(state.items, renderItem, () => {});
    } else {
      rootEl.style.height = `${state.items.length * estimatedItemHeight}px`;
      computeVisible();
      renderVisible();
    }
  }

  function destroy() {
    if (observer) { observer.disconnect(); observer = null; }
    if (rootEl) { rootEl.remove(); rootEl = null; }
    state = { items: [], scrollEl: null, start: 0, end: 0 };
  }

  return { mount, update, destroy };
}
```
